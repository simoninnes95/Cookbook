# Use this ubuntu debugging pod manifest:
```yaml
apiVersion: v1
kind: Pod
metadata:
  name: ubuntu
spec:
  nodeSelector:
    cloud.google.com/gke-nodepool: db-pool  # This forces the pod onto the right pool
  containers:
  - name: ubuntu
    image: ubuntu:latest
    command: ["sleep", "86400"]
    resources:
      limits:
        cpu: "2"
        memory: "4Gi"
      requests:
        cpu: "1"
        memory: "2Gi"
    volumeMounts:
    - name: ssd-volume
      mountPath: /data
  volumes:
  - name: ssd-volume
    hostPath:
      path: /mnt/stateful_partition/kube-ephemeral-ssd
      type: Directory
```

# Create a debugging pod:


    $ kubectl create -f ubuntu.yaml => use this when iterating quickly $ kubectl replace --force -f ubuntu.yaml -n load-test-2899
        pod/ubuntu created

    $ kubectl exec ubuntu -it -n <namespace> -- bash

    $ kubectl debug node/<node-name> -it --image=ubuntu
